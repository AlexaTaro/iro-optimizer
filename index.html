function generate() { const lang = document.querySelector('.lang-btn.active').innerText; // 現在の選択言語を取得 const g = document.getElementById('garlic'); const v = document.getElementById('vegetable'); const f = document.getElementById('fat'); const s = document.getElementById('salty'); // 選択された言語に応じたデータ属性から値を取得 const getVal = (el) => { const opt = el.options[el.selectedIndex]; if (lang === 'EN') return opt.getAttribute('data-en') || opt.value; if (lang === 'ZH') return opt.getAttribute('data-zh') || opt.value; return opt.value; }; const gVal = getVal(g); const vVal = getVal(v); const fVal = getVal(f); const sVal = (lang === 'JA') ? s.value : (s.value.trim() ? "Salty" : ""); // カラメの多言語対応 // 全マシ判定 const isAll = (g.value.includes("ニンニク") && v.value.includes("ヤサイ") && f.value.includes("アブラ") && s.value.includes("カラメ")); let res = ""; let sub = ""; // 日本語の読み/補足用 if (isAll) { if (lang === 'EN') { res = "ZEN-MASHI"; sub = "All Toppings"; } else if (lang === 'ZH') { res = "全加 (Quán jiā)"; sub = "全部トッピング"; } else { res = "全マシ"; sub = "ZEN-MASHI"; } } else { res = [gVal, vVal, fVal, sVal].filter(x => x && x.trim() !== "").join(" "); } if (!res || res.includes("なし") || res.includes("None")) { res = (lang === 'EN') ? "SONO-MAMA" : (lang === 'ZH' ? "就这样 (Jiù zhèyàng)" : "そのままで"); sub = "Default / No extra"; } // 表示の更新 document.getElementById('call-output').innerHTML = `
Voice Call:
` + `
${res}！
` + `
${sub}
`; document.getElementById('sys-msg').innerText = `> STATUS: 200 OK (${lang} Mode)`; }

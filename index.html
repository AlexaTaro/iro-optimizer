function generate() { const g = document.getElementById('garlic').value; const v = document.getElementById('vegetable').value; const f = document.getElementById('fat').value; const s = document.getElementById('salty').value; const isZenMashi = (g === "ニンニク" && v === "ヤサイ" && f === "アブラ" && s === "カラメ"); const isZenMashimashi = (g === "ニンニクマシマシ" && v === "ヤサイマシマシ" && f === "アブラマシマシ" && s === "カラメ"); let res = ""; let reading = ""; // 発音ガイド用 if (isZenMashimashi) { res = "全マシマシ"; reading = "ZEN-MASHI-MASHI"; } else if (isZenMashi) { res = "全マシ"; reading = "ZEN-MASHI"; } else { res = [g, v, f, s].filter(x => x.trim() !== "").join(""); // 読みの生成（簡易変換） reading = res .replace(/ニンニク/g, "NINNIKU ") .replace(/ヤサイ/g, "YASAI ") .replace(/アブラ/g, "ABURA ") .replace(/カラメ/g, "KARAME ") .replace(/マシマシ/g, "MASHI-MASHI ") .replace(/マシ/g, "MASHI ") .replace(/少なめ/g, "SUKUNAME ") .replace(/抜き/g, "NUKI "); } if (res === "" || res === "なし") { res = "そのままで"; reading = "SONO-MAMA-DE"; } // 表示の更新 document.getElementById('call-output').innerHTML = `
Voice Call:
` + `
${res}！
` + `
[ ${reading.trim()} ]
`; document.getElementById('sys-msg').innerText = "> STATUS: 200 OK (Phonetic Output Ready)"; }
